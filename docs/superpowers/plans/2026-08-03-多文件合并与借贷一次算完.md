# src/core/direction_resolver.py
# -*- coding: utf-8 -*-
"""方向判断器 - 三维度综合判断科目借贷方向"""
from __future__ import annotations


class DirectionResolver:
    """按优先级判断科目方向：余额 > 编码首位 > 名称关键词 > 默认"""

    # 长词在前，短词在后，确保"其他应收"优先于"应收"匹配
    DEBIT_KEYWORDS = [
        "其他应收", "应收账款", "应收", "预付账款", "预付",
        "待摊费用", "长期待摊", "待摊",
    ]
    CREDIT_KEYWORDS = [
        "其他应付", "应付账款", "应付", "预收账款", "预收",
        "预提费用", "预提",
    ]

    def resolve(
        self,
        subject_code: str,
        subject_name: str,
        opening_debit_total: float,
        opening_credit_total: float,
        default_direction: str,
    ) -> tuple[str, str]:
        """返回 (direction, basis)，basis 为判断依据"""

        # 第一优先：期初余额方向
        if opening_debit_total > opening_credit_total:
            return "debit", "balance"
        if opening_credit_total > opening_debit_total:
            return "credit", "balance"

        # 第二优先：科目编码首位
        code = str(subject_code).strip()
        if code:
            first = code[0]
            if first == "1":
                return "debit", "code"
            if first == "2":
                return "credit", "code"

        # 第三优先：科目名称关键词（长词优先）
        name = str(subject_name or "")
        for keyword in self.DEBIT_KEYWORDS:
            if keyword in name:
                return "debit", "name"
        for keyword in self.CREDIT_KEYWORDS:
            if keyword in name:
                return "credit", "name"

        # 兜底：默认方向
        return default_direction, "default"
```

def merge_journal_files(dfs: list[pd.DataFrame]) -> pd.DataFrame:
    """合并多个序时账 DataFrame，每个自动匹配列名后拼接"""
    from src.core.column_mapper import ColumnMapper, REQUIRED_FIELDS
    from src.core.data_loader import DataLoader

    loader = DataLoader()
    normalized = []
    required = REQUIRED_FIELDS["journal"]

    for i, df in enumerate(dfs):
        mapper = ColumnMapper()
        mapper.auto_map(list(df.columns), field_type="journal")
        # 校验必要字段
        mapping = mapper.get_mapping()
        missing = [f for f in required if not mapping.get(f)]
        if missing:
            raise ValueError(f"第 {i+1} 个文件缺少必要字段: {', '.join(missing)}")
        # 应用映射统一列名
        normalized_df = loader.apply_mapping(df, mapping)
        normalized.append(normalized_df)

    return pd.concat(normalized, ignore_index=True)
```

- `askopenfilename` 改为 `askopenfilenames`
- 读取所有文件，调 `merge_journal_files`
- 预览显示"已合并 N 个文件，共 M 行"
